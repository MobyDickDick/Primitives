# Outputs: c0..c3, err; a=0 -> [0,1].
# Inputs: a0, a1, a2, a3

g1 = NAND(a0, g0)
g2 = NAND(g1, g1)

g6 = NAND(a2, a2)
g7 = NAND(a3, a3)
g8 = NAND(g6, g7)
g9 = NAND(g8, g8)
g10 = NAND(a3, g8)
g11 = NAND(a1, g9)
g12 = NAND(g10, g11)
g13 = NAND(g5, g9)
g14 = NAND(g13, g13)
c0 = g12
c1 = g8
c2 = g2
c3 = g2
err = g14
